\section{Program flowchart}
\label{chapter3}

Figure~\ref{fig:flow_esp32} shows the program flow of the ESP32 sensor node and Figure~\ref{fig:flow_rpi} the program flow of the Raspberry Pi gateway. Hazard handling (H1--H3, see Chapter~\ref{chapter4}) and critical intake results are highlighted in red.

% Flowchart ESP32 (sensor node)

\begin{figure}[H]
  \centering
  \resizebox{!}{0.82\textheight}{%
\begin{tikzpicture}[
    font=\footnotesize,
    node distance=6mm and 8mm,
    base/.style={draw, align=center, minimum height=8mm, text width=38mm},
    term/.style={base, rounded corners=4mm, fill=gray!15, text width=28mm},
    proc/.style={base},
    dec/.style={draw, diamond, aspect=2.4, align=center, inner sep=1pt, text width=20mm},
    haz/.style={base, fill=red!8, draw=red!70!black, text width=46mm},
    arr/.style={-{Stealth[length=2mm]}},
    lbl/.style={font=\scriptsize, inner sep=1pt}
  ]
  % --- Start-up ---
  \node[term] (start) {Start ESP32\\(reset button)};

  % --- Heartbeat thread (runs independently, Req 3.4) ---
  \node[term, right=40mm of start, text width=40mm] (tstart) {Heartbeat thread\\(after schedule received)};
  \node[dec, below=10mm of tstart] (tcon) {Connected?};
  \node[proc, below=of tcon, text width=36mm] (thb) {Send heartbeat to RPi};
  \node[proc, below=of thb, text width=36mm] (twait) {Wait 30\,s};
  \node[dec, below=of twait, text width=24mm] (tack) {Ack from RPi\\$\leq$ 2\,min ago?};
  \node[haz, anchor=north west, text width=44mm] (trec) at ($(tcon.east)+(10mm,3mm)$) {\textbf{H3:} reconnect (ECDHE + PSK),\\wait 5\,s; on success: send\\buffered results, receive time};
  \node[haz, anchor=west, text width=44mm] (tlost) at ($(tack.east)+(10mm,0)$) {\textbf{H3:} connection lost:\\disconnect, autonomous mode\\(evaluate, buffer results)};
  \node[proc, below=of start] (init) {Initialise hardware\\(GPIO, HX711, timer, threads)};
  \node[proc, below=of init] (led) {LED self-test (R, G, B)};

  % --- Bluetooth (H3) ---
  \node[proc, below=of led] (ble) {Establish BLE connection\\(ECDHE + PSK)};
  \node[dec, below=of ble] (d2) {Connected?};
  \node[dec, left=15mm of d2] (c2) {Attempt\\$< 3$?};
  \node[haz, below=of c2] (h3c) {\textbf{H3:} no schedule and time,\\no evaluation, red LED\\flashes, retry connection};
  \node[proc, below=of d2] (rx) {Receive schedule\\and current time};

  % --- Sensor check at start-up (H2) ---
  \node[proc, below=10mm of rx] (tare) {Load calibration (NVS),\\read HX711};
  \node[dec, below=of tare] (d1) {Values\\plausible?};
  \node[dec, right=15mm of d1] (c1) {Repeated\\$< 2\times$?};
  \node[haz, below=of c1] (h2a) {\textbf{H2:} no evaluation,\\red LED flashes, report fault\\to RPi, recheck every 10\,s};

  % --- Filling check at start-up ---
  \node[dec, below=24mm of d1] (dfill) {Filling\\correct?};
  \node[proc, right=15mm of dfill, text width=44mm] (hfill) {Red LED flashes, report\\incorrect filling to RPi,\\recheck every 10\,s};

  % --- Main loop (table of expected weights) ---
  \node[proc, below=of dfill, text width=46mm] (table) {Create table of expected weights\\(from, to, $W_{before}$, $W_{after}$)};
  \node[proc, below=of table] (meas) {Measure weight $W$, select\\row for current time};
  \node[dec, below=of meas] (d3) {Value\\plausible?};
  \node[dec, right=15mm of d3] (c3) {Repeated\\$< 2\times$?};
  \node[haz, below=of c3] (h2b) {\textbf{H2:} suspend evaluation,\\red LED flashes, report fault\\to RPi, recheck every 10\,s};
  \node[dec, below=26mm of d3] (dafter) {$W \approx W_{after}$?};
  \node[proc, right=15mm of dafter, text width=34mm, fill=green!8, draw=green!50!black] (green) {LED green\\(dose taken)};
  \node[dec, below=of dafter] (dint) {Intake\\window?};
  \node[proc, left=12mm of dint, text width=40mm, fill=red!8, draw=red!70!black] (miss) {LED red (missed dose or\\unscheduled removal)};
  \node[dec, below=of dint] (dbefore) {$W \approx W_{before}$?};
  \node[proc, right=15mm of dbefore, text width=34mm] (wait) {LED off (not taken\\yet), keep waiting};
  \node[proc, below=of dbefore, fill=red!8, draw=red!70!black] (dev) {LED red / blue\\(wrong amount)};
  \node[proc, below=of dev, text width=48mm] (send) {Report each new result once to\\RPi (if not connected: buffer,\\see heartbeat thread)};
  \node[dec, below=of send] (dday) {Last row\\ended?};
  \node[dec, below=of dday] (dlast) {$W \approx W_{after}$?};
  \node[proc, right=15mm of dlast, text width=40mm, fill=red!8, draw=red!70!black] (lastmiss) {LED red, report missed\\dose to RPi};
  \node[proc, below=of dlast, text width=20mm] (lastoff) {LED off};
  \node[proc, below=of lastoff] (dayend) {Report end of day to RPi};
  \node[term, below=of dayend] (dend) {Deep sleep until\\reset button};

  % --- Arrows: heartbeat thread ---
  \coordinate (tin) at ($(tstart.south)!0.5!(tcon.north)$);
  \draw[arr] (tstart) -- (tcon);
  \draw[arr] (tcon) -- node[lbl, right]{yes} (thb);
  \draw[arr] (thb) -- (twait);
  \draw[arr] (twait) -- (tack);
  \draw[arr] (tcon.east) -- node[lbl, above]{no} (tcon.east -| trec.west);
  \draw[arr] (trec.north) |- (tin);
  \draw[arr] (tack) -- node[lbl, above]{no} (tlost);
  \draw[arr] (tlost.north) -- (tlost.north |- trec.south);
  \draw[arr] (tack.west) -- node[lbl, above]{yes} ++(-6mm,0) |- (tin);

  % --- Arrows: start-up and Bluetooth ---
  \draw[arr] (start) -- (init);
  \draw[arr] (init) -- (led);
  \draw[arr] (led) -- (ble);
  \draw[arr] (ble) -- (d2);
  \draw[arr] (d2) -- node[lbl, above]{no} (c2);
  \draw[arr] (c2.north) |- node[lbl, pos=0.25, left]{yes, wait 5\,s} (ble.west);
  \draw[arr] (c2) -- node[lbl, right]{no} (h3c);
  \draw[arr] (h3c.west) -- ++(-5mm,0) |- ($(led.south)!0.5!(ble.north)$);
  \draw[arr] (d2) -- node[lbl, right]{yes} (rx);
  \draw[arr] (rx) -- (tare);

  % --- Arrows: sensor check at start-up ---
  \draw[arr] (tare) -- (d1);
  \draw[arr] (d1) -- node[lbl, above]{no} (c1);
  \draw[arr] (c1.north) |- node[lbl, pos=0.25, right]{yes, poll again} (tare.east);
  \draw[arr] (c1) -- node[lbl, right]{no} (h2a);
  \draw[arr] (h2a.east) -- ++(4mm,0) |- ($(tare.east)+(0,2mm)$);
  \draw[arr] (d1) -- node[lbl, right]{yes} (dfill);
  \draw[arr] (dfill) -- node[lbl, above]{no} (hfill);
  \draw[arr] (hfill.east) -- ++(10mm,0) |- ($(tare.east)+(0,4mm)$);
  \draw[arr] (dfill) -- node[lbl, right]{yes} (table);

  % --- Arrows: main loop ---
  \draw[arr] (table) -- (meas);
  \draw[arr] (meas) -- (d3);
  \draw[arr] (d3) -- node[lbl, above]{no} (c3);
  \draw[arr] (c3.north) |- node[lbl, pos=0.25, right]{yes, poll again} (meas.east);
  \draw[arr] (c3) -- node[lbl, right]{no} (h2b);
  \draw[arr] (h2b.east) -- ++(4mm,0) |- ($(meas.east)+(0,2mm)$);
  \draw[arr] (d3) -- node[lbl, right]{yes} (dafter);
  \draw[arr] (dafter) -- node[lbl, above]{yes} (green);
  \draw[arr] (dafter) -- node[lbl, right]{no} (dint);
  \draw[arr] (dint) -- node[lbl, above]{no} (miss);
  \draw[arr] (dint) -- node[lbl, right]{yes} (dbefore);
  \draw[arr] (dbefore) -- node[lbl, above]{yes} (wait);
  \draw[arr] (dbefore) -- node[lbl, right]{no} (dev);
  \draw[arr] (dev) -- (send);
  \draw (green.east) -- ++(5mm,0) coordinate (gx);
  \draw[arr] (gx) |- ($(send.east)+(0,2mm)$);
  \draw[arr] (miss.south) |- (send.west);
  \draw[arr] (wait.east) -- ++(12mm,0) |- (dday.east);
  \draw[arr] (send) -- (dday);
  \draw[arr] (dday) -- node[lbl, right]{yes} (dlast);
  \draw[arr] (dlast) -- node[lbl, right]{yes} (lastoff);
  \draw[arr] (dlast) -- node[lbl, above]{no} (lastmiss);
  \draw[arr] (lastoff) -- (dayend);
  \draw[arr] (lastmiss.south) |- (dayend.east);
  \draw[arr] (dayend) -- (dend);
  \draw[arr] (dday.west) -- node[lbl, above, pos=0.9]{no} ++(-112mm,0) coordinate (lp) |- (meas.west);
\end{tikzpicture}
  }
  \caption{Program flowchart ESP32 (sensor node)}
  \label{fig:flow_esp32}
\end{figure}

% Flowchart Raspberry Pi (gateway)

\begin{figure}[H]
  \centering
  \resizebox{!}{0.95\textheight}{%
\begin{tikzpicture}[
    font=\footnotesize,
    node distance=6mm and 8mm,
    base/.style={draw, align=center, minimum height=8mm, text width=46mm},
    term/.style={base, rounded corners=4mm, fill=gray!15, text width=36mm},
    proc/.style={base},
    dec/.style={draw, diamond, aspect=2.4, align=center, inner sep=1pt, text width=28mm},
    haz/.style={base, fill=red!8, draw=red!70!black, text width=52mm},
    arr/.style={-{Stealth[length=2mm]}},
    lbl/.style={font=\scriptsize, inner sep=1pt}
  ]
  % --- Start-up and time (H1) ---
  \node[term] (start) {Start Raspberry Pi};
  \node[proc, below=of start] (init) {Initialise threads,\\BlueZ, libcurl};
  \node[proc, below=of init] (ntp) {Synchronise time via NTP};
  \node[dec, below=of ntp] (dn) {Time\\valid?};
  \node[dec, right=15mm of dn] (cn) {Attempt\\$< 3$?};
  \node[haz, below=of cn] (h1) {\textbf{H1:} Telegram alert ``no valid\\time'', stop operation (no\\schedule, time or reminders)};
  \node[proc, below=of h1] (w5) {Wait 5\,min, retry NTP};
  \node[dec, below=of w5] (dn5) {Time\\valid?};

  % --- Bluetooth (H3) ---
  \node[proc, anchor=north] (load) at ($(dn.south |- dn5.south)+(0,-8mm)$) {Load schedule from config\\file, calculate target weights};
  \node[proc, below=of load] (ble) {Connect to stored ESP32\\via BLE (ECDHE + PSK)};
  \node[dec, below=of ble] (db) {Connected?};
  \node[dec, right=15mm of db] (cb) {No connection\\for 2\,min?};
  \node[haz, below=of cb] (h3) {\textbf{H3:} Telegram alert (once)\\``dispenser not reachable'',\\reconnect in background (on\\reconnect: send schedule\\and time), continue};
  \node[proc, below=of db] (tx) {Send today's schedule and\\current time to ESP32};

  % --- Main loop ---
  \node[proc, anchor=north] (wait) at ($(tx.south |- h3.south)+(0,-14mm)$) {Wait for event\\(timer, BLE)};
  \node[dec, below=of wait] (dsync) {NTP re-sync\\due (60\,min)?};
  \node[proc, right=15mm of dsync, text width=40mm] (resync) {Synchronise time\\via NTP (on failure: H1)};
  \node[dec, below=of dsync] (dhb) {Heartbeat\\$\leq$ 2\,min ago\\or day ended?};
  \node[haz, right=15mm of dhb] (h3b) {\textbf{H3:} ESP32 without power\\or BLE lost: Telegram alert\\(once), reconnect in background};
  \node[dec, below=16mm of dhb] (ddue) {Intake time\\reached?};
  \node[proc, right=15mm of ddue, text width=40mm] (tg) {Telegram reminder\\(medications and quantity)};
  \node[dec, below=of ddue] (dres) {Message from\\ESP32?};
  \node[proc, below=of dres] (ack) {If heartbeat: send\\acknowledgement to ESP32};
  \node[dec, below=of ack] (dbad) {Red / blue or\\fault reported?};
  \node[proc, right=15mm of dbad, text width=40mm] (warn) {Telegram warning};

  % --- Arrows: start-up and time ---
  \draw[arr] (start) -- (init);
  \draw[arr] (init) -- (ntp);
  \draw[arr] (ntp) -- (dn);
  \draw[arr] (dn) -- node[lbl, above]{no} (cn);
  \draw[arr] (cn.north) |- node[lbl, pos=0.25, right]{yes, wait 30\,s} (ntp.east);
  \draw[arr] (cn) -- node[lbl, right]{no} (h1);
  \draw[arr] (h1) -- (w5);
  \draw[arr] (w5) -- (dn5);
  \draw[arr] (dn5.east) -- node[lbl, above]{no} ++(4mm,0) |- (w5.east);
  \draw[arr] (dn5.west) -- node[lbl, above]{yes} (dn5.west -| dn.south);
  \draw[arr] (dn) -- node[lbl, right]{yes} (load);

  % --- Arrows: Bluetooth ---
  \draw[arr] (load) -- (ble);
  \draw[arr] (ble) -- (db);
  \draw[arr] (db) -- node[lbl, above]{no} (cb);
  \draw[arr] (cb.north) |- node[lbl, pos=0.25, right]{no, wait 5\,s} (ble.east);
  \draw[arr] (cb) -- node[lbl, right]{yes} (h3);
  \draw[arr] (h3.south) |- ($(tx.south |- h3.south)+(0,-6mm)$);
  \draw[arr] (db) -- node[lbl, right]{yes} (tx);
  \draw[arr] (tx) -- (wait);

  % --- Arrows: main loop ---
  \draw[arr] (wait) -- (dsync);
  \draw[arr] (dsync) -- node[lbl, above]{yes} (resync);
  \draw[arr] (resync.north) |- (wait.east);
  \draw[arr] (dsync) -- node[lbl, right]{no} (dhb);
  \draw[arr] (dhb) -- node[lbl, above]{no} (h3b);
  \draw[arr] (h3b.south) |- ($(dhb.south)!0.5!(ddue.north)$);
  \draw[arr] (dhb) -- node[lbl, right, pos=0.2]{yes} (ddue);
  \draw[arr] (ddue) -- node[lbl, above]{yes} (tg);
  \draw[arr] (tg) |- (dres);
  \draw[arr] (ddue) -- node[lbl, right]{no} (dres);
  % common loop line on the left, below the last diamond
  \coordinate (lx) at ($(dhb.west)+(-10mm,0)$);
  \coordinate (wy) at ($(dbad.south)+(0,-6mm)$);
  \draw (dres.west) -- node[lbl, above]{no} (dres.west -| lx);
  \draw[arr] (dres) -- node[lbl, right]{yes} (ack);
  \draw[arr] (ack) -- (dbad);
  \draw[arr] (dbad) -- node[lbl, above]{yes} (warn);
  \draw (dbad.west) -- node[lbl, above]{no} (dbad.west -| lx);
  \draw (warn.south) |- (wy -| lx);
  \draw[arr] (wy -| lx) |- (wait.west);
\end{tikzpicture}
  }
  \caption{Program flowchart Raspberry Pi (gateway)}
  \label{fig:flow_rpi}
\end{figure}
